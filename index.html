    <script>

        const entryTypeSelect = document.getElementById('entryType');

        const retailerGroup = document.getElementById('retailerGroup');

        const categorySelect = document.getElementById('category');

        const wattageGroup = document.getElementById('wattageGroup');

        const stockForm = document.getElementById('stockForm');

        const inventoryTable = document.getElementById('inventoryTable');


        // Toggle Retailer Field

        entryTypeSelect.addEventListener('change', (e) => {

            if (e.target.value === 'Dispatched') {

                retailerGroup.style.display = 'block';

            } else {

                retailerGroup.style.display = 'none';

            }

        });


        // Toggle Wattage Dropdown for Bulbs

        function handleCategoryChange() {

            const val = categorySelect.value;

            if (val === 'Light Bulb H1' || val === 'Light Bulb H4') {

                wattageGroup.style.display = 'block';

            } else {

                wattageGroup.style.display = 'none';

            }

        }


        // Save & Render Inventory using LocalStorage

        let records = JSON.parse(localStorage.getItem('auto_addiction_inventory')) || [];


        function renderTable() {

            inventoryTable.innerHTML = '';

            if(records.length === 0) {

                inventoryTable.innerHTML = '<tr><td colspan="5" style="text-align:center; color: var(--text-muted);">No items recorded yet.</td></tr>';

                return;

            }


            records.forEach((item, index) => {

                const tr = document.createElement('tr');

                tr.innerHTML = `

                    <td><span class="badge" style="color: ${item.type === 'In Stock' ? '#4cc9f0' : '#f72585'}">${item.type}</span></td>

                    <td><strong>${item.category}</strong> ${item.wattage ? `(${item.wattage})` : ''}</td>

                    <td>${item.quantity}</td>

                    <td>${item.retailer || '-'}</td>

                    <td><button class="btn-del" onclick="deleteEntry(${index})">X</button></td>

                `;

                inventoryTable.appendChild(tr);

            });

        }


        stockForm.addEventListener('submit', (e) => {

            e.preventDefault();

            const type = entryTypeSelect.value;

            const category = categorySelect.value;

            const wattage = (category === 'Light Bulb H1' || category === 'Light Bulb H4') ? document.getElementById('wattage').value : '';

            const quantity = document.getElementById('quantity').value;

            const retailer = type === 'Dispatched' ? document.getElementById('retailerName').value : '';


            if(!category) return alert('Please select a product category');


            const entry = { type, category, wattage, quantity, retailer };

            records.push(entry);

            localStorage.setItem('auto_addiction_inventory', JSON.stringify(records));

            

            stockForm.reset();

            wattageGroup.style.display = 'none';

            retailerGroup.style.display = 'none';

            renderTable();

        });


        function deleteEntry(index) {

            records.splice(index, 1);

            localStorage.setItem('auto_addiction_inventory', JSON.stringify(records));

            renderTable();

        }


        // Initial Load

        renderTable();

    </script>
